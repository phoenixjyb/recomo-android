cmake_minimum_required(VERSION 3.22.1)
project(openvins_jni LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(openvins_jni SHARED
    openvins_jni.cpp
)

find_library(log-lib log)

target_link_libraries(openvins_jni
    ${log-lib}
)

if (DEFINED OPENVINS_ROOT)
    set(OPENVINS_ROOT "${OPENVINS_ROOT}")
else()
    set(OPENVINS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../external/open_vins")
endif()
if (DEFINED OPENVINS_EIGEN_ROOT)
    set(OPENVINS_EIGEN_ROOT "${OPENVINS_EIGEN_ROOT}")
else()
    set(OPENVINS_EIGEN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../external/eigen")
endif()
if (DEFINED OPENVINS_OPENCV_DIR)
    set(OPENVINS_OPENCV_DIR "${OPENVINS_OPENCV_DIR}")
else()
    set(OPENVINS_OPENCV_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../external/opencv-mobile/prebuilt/sdk/native/jni")
endif()
if (DEFINED OPENVINS_BOOST_ROOT)
    set(OPENVINS_BOOST_ROOT "${OPENVINS_BOOST_ROOT}")
else()
    set(OPENVINS_BOOST_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../external/Boost-for-Android/build")
endif()
option(OPENVINS_ENABLE "Enable OpenVINS native build" OFF)
option(OPENVINS_DEPS_ONLY "Enable OpenVINS dependency smoke test" OFF)
option(OPENVINS_LINK_OPENCV "Link OpenCV static libs for JNI" OFF)

set(OPENVINS_DEPS_AVAILABLE OFF)
if (OPENVINS_DEPS_ONLY OR OPENVINS_ENABLE)
    if (EXISTS "${OPENVINS_EIGEN_ROOT}/cmake/FindEigen3.cmake")
        list(APPEND CMAKE_MODULE_PATH "${OPENVINS_EIGEN_ROOT}/cmake")
        set(EIGEN3_INCLUDE_DIR "${OPENVINS_EIGEN_ROOT}")
    endif()
    # Use ABI-specific OpenCV config for Android
    if (ANDROID AND EXISTS "${OPENVINS_OPENCV_DIR}/abi-${ANDROID_ABI}/OpenCVConfig.cmake")
        set(OpenCV_DIR "${OPENVINS_OPENCV_DIR}/abi-${ANDROID_ABI}")
    elseif (EXISTS "${OPENVINS_OPENCV_DIR}/OpenCVConfig.cmake")
        set(OpenCV_DIR "${OPENVINS_OPENCV_DIR}")
    endif()
    if (EXISTS "${OPENVINS_BOOST_ROOT}/include")
        set(BOOST_ROOT "${OPENVINS_BOOST_ROOT}")
        set(BOOST_INCLUDEDIR "${OPENVINS_BOOST_ROOT}/include")
        set(BOOST_LIBRARYDIR "${OPENVINS_BOOST_ROOT}/lib")
        set(Boost_NO_SYSTEM_PATHS ON)
        set(Boost_USE_STATIC_LIBS ON)
        set(Boost_USE_STATIC_RUNTIME OFF)
    endif()

    set(_openvins_missing "")
    if (NOT EXISTS "${OPENVINS_EIGEN_ROOT}/Eigen")
        list(APPEND _openvins_missing "Eigen")
    endif()
    if (NOT EXISTS "${OPENVINS_OPENCV_DIR}/OpenCVConfig.cmake")
        list(APPEND _openvins_missing "OpenCV")
    endif()
    if (NOT EXISTS "${OPENVINS_BOOST_ROOT}/include")
        list(APPEND _openvins_missing "Boost")
    endif()

    if (_openvins_missing)
        message(WARNING "OpenVINS deps missing: ${_openvins_missing}")
    else()
        find_package(OpenCV QUIET)
        find_package(Boost QUIET COMPONENTS filesystem system)
        if (NOT Boost_FOUND)
            file(GLOB _boost_include_dirs "${OPENVINS_BOOST_ROOT}/include/boost-*")
            file(GLOB _boost_fs_libs "${OPENVINS_BOOST_ROOT}/lib/libboost_filesystem*.a")
            file(GLOB _boost_sys_libs "${OPENVINS_BOOST_ROOT}/lib/libboost_system*.a")
            if (_boost_include_dirs AND _boost_fs_libs AND _boost_sys_libs)
                list(GET _boost_include_dirs 0 _boost_include_guess)
                list(GET _boost_fs_libs 0 BOOST_FILESYSTEM_LIB)
                list(GET _boost_sys_libs 0 BOOST_SYSTEM_LIB)
                set(Boost_FOUND TRUE)
                set(Boost_INCLUDE_DIRS "${_boost_include_guess}")
                set(Boost_LIBRARIES "${BOOST_FILESYSTEM_LIB}" "${BOOST_SYSTEM_LIB}")
            endif()
            unset(_boost_include_dirs)
            unset(_boost_fs_libs)
            unset(_boost_sys_libs)
            unset(_boost_include_guess)
        endif()
        if (OpenCV_FOUND AND Boost_FOUND)
            set(OPENVINS_DEPS_AVAILABLE ON)
        else()
            if (NOT OpenCV_FOUND)
                message(WARNING "OpenCV not found via ${OPENVINS_OPENCV_DIR}")
            endif()
            if (NOT Boost_FOUND)
                message(WARNING "Boost filesystem/system not found via ${OPENVINS_BOOST_ROOT}")
            endif()
        endif()
    endif()
endif()

if (OPENVINS_ENABLE)
    set(OPENVINS_LINK_OPENCV ON)
endif()

if (OPENVINS_DEPS_AVAILABLE)
    target_compile_definitions(openvins_jni PRIVATE OPENVINS_DEPS_AVAILABLE=1)
    if (DEFINED EIGEN3_INCLUDE_DIR)
        target_include_directories(openvins_jni PRIVATE "${EIGEN3_INCLUDE_DIR}")
    endif()
    if (OpenCV_FOUND)
        target_include_directories(openvins_jni PRIVATE ${OpenCV_INCLUDE_DIRS})
        if (OPENVINS_LINK_OPENCV)
            if (OpenCV_LIBS)
                target_link_libraries(openvins_jni ${OpenCV_LIBS})
            elseif (OpenCV_LIBRARIES)
                target_link_libraries(openvins_jni ${OpenCV_LIBRARIES})
            endif()
        endif()
    endif()
    if (Boost_FOUND)
        target_include_directories(openvins_jni PRIVATE ${Boost_INCLUDE_DIRS})
        target_link_libraries(openvins_jni ${Boost_LIBRARIES})
    endif()

    if (OpenCV_FOUND AND OPENVINS_LINK_OPENCV)
        if (DEFINED ANDROID_NDK)
            set(_ndk_root "${ANDROID_NDK}")
        elseif (DEFINED CMAKE_ANDROID_NDK)
            set(_ndk_root "${CMAKE_ANDROID_NDK}")
        endif()
        if (DEFINED _ndk_root AND EXISTS "${_ndk_root}/toolchains/llvm/prebuilt")
            file(GLOB _ndk_prebuilts "${_ndk_root}/toolchains/llvm/prebuilt/*")
            list(LENGTH _ndk_prebuilts _ndk_prebuilt_count)
            if (_ndk_prebuilt_count GREATER 0)
                list(GET _ndk_prebuilts 0 _ndk_prebuilt)
                file(GLOB _clang_dirs "${_ndk_prebuilt}/lib/clang/*")
                list(SORT _clang_dirs)
                list(LENGTH _clang_dirs _clang_dir_count)
                if (_clang_dir_count GREATER 0)
                    list(GET _clang_dirs -1 _clang_dir)
                    if (CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
                        set(_omp_arch "aarch64")
                    elseif (CMAKE_ANDROID_ARCH_ABI STREQUAL "armeabi-v7a")
                        set(_omp_arch "arm")
                    elseif (CMAKE_ANDROID_ARCH_ABI STREQUAL "x86")
                        set(_omp_arch "i386")
                    elseif (CMAKE_ANDROID_ARCH_ABI STREQUAL "x86_64")
                        set(_omp_arch "x86_64")
                    endif()
                    if (DEFINED _omp_arch)
                        set(_omp_candidate "${_clang_dir}/lib/linux/${_omp_arch}/libomp.a")
                        if (EXISTS "${_omp_candidate}")
                            target_link_libraries(openvins_jni "${_omp_candidate}")
                        endif()
                    endif()
                endif()
            endif()
        endif()
        unset(_ndk_root)
        unset(_ndk_prebuilts)
        unset(_ndk_prebuilt_count)
        unset(_ndk_prebuilt)
        unset(_clang_dirs)
        unset(_clang_dir_count)
        unset(_clang_dir)
        unset(_omp_arch)
        unset(_omp_candidate)
    endif()
endif()

if (OPENVINS_ENABLE AND EXISTS "${OPENVINS_ROOT}/ov_core/CMakeLists.txt")
    message(STATUS "OpenVINS source detected at ${OPENVINS_ROOT}")
    set(ENABLE_ROS OFF CACHE BOOL "Disable ROS for Android build" FORCE)
    set(ENABLE_ARUCO_TAGS OFF CACHE BOOL "Disable ArUco tags for Android" FORCE)

    if (NOT OPENVINS_DEPS_AVAILABLE)
        message(FATAL_ERROR "OpenVINS deps missing. Enable OPENVINS_DEPS_ONLY to debug.")
    endif()

    # Ensure OpenCV_DIR and Eigen paths are visible to child CMakeLists
    if (ANDROID)
        set(_opencv_config_dir "${OPENVINS_OPENCV_DIR}/abi-${ANDROID_ABI}")
    else()
        set(_opencv_config_dir "${OPENVINS_OPENCV_DIR}")
    endif()
    # Must set BOTH as cache and as regular variable for child scope
    set(OpenCV_DIR "${_opencv_config_dir}" CACHE PATH "OpenCV config dir" FORCE)
    set(OpenCV_DIR "${_opencv_config_dir}" PARENT_SCOPE)
    set(CMAKE_PREFIX_PATH "${_opencv_config_dir};${CMAKE_PREFIX_PATH}" CACHE STRING "Prefix path" FORCE)
    message(STATUS "Setting OpenCV_DIR to ${_opencv_config_dir}")
    
    set(EIGEN3_INCLUDE_DIR "${OPENVINS_EIGEN_ROOT}" CACHE PATH "Eigen include dir" FORCE)
    list(APPEND CMAKE_MODULE_PATH "${OPENVINS_EIGEN_ROOT}/cmake")
    set(CMAKE_MODULE_PATH "${CMAKE_MODULE_PATH}" CACHE STRING "Module path" FORCE)

    # Set ov_core source path for ov_init (it needs to include ov_core headers)
    set(OV_CORE_SRC_DIR "${OPENVINS_ROOT}/ov_core/src" CACHE PATH "ov_core source dir" FORCE)
    set(OV_INIT_SRC_DIR "${OPENVINS_ROOT}/ov_init/src" CACHE PATH "ov_init source dir" FORCE)

    # Add OpenVINS subdirectories
    add_subdirectory(${OPENVINS_ROOT}/ov_core ${CMAKE_BINARY_DIR}/openvins_ov_core)
    add_subdirectory(${OPENVINS_ROOT}/ov_init ${CMAKE_BINARY_DIR}/openvins_ov_init)
    add_subdirectory(${OPENVINS_ROOT}/ov_msckf ${CMAKE_BINARY_DIR}/openvins_ov_msckf)
    
    # Enable OPENVINS_ENABLE flag for JNI
    target_compile_definitions(openvins_jni PRIVATE OPENVINS_ENABLE=1)
    
    # Include OpenVINS headers
    target_include_directories(openvins_jni PRIVATE 
        ${OPENVINS_ROOT}/ov_core/src
        ${OPENVINS_ROOT}/ov_init/src
        ${OPENVINS_ROOT}/ov_msckf/src
    )
    
    # Link OpenVINS libraries
    target_link_libraries(openvins_jni 
        ov_core_lib
        ov_init_lib
        ov_msckf_lib
    )
endif()
